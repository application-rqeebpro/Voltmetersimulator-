// هذا المكون جاهز للاستخدام في أي مشروع React / Next.js / Vite

import React, { useState, useEffect, useRef } from 'react';

export default function MultimeterSimulator() {
  const [isOn, setIsOn] = useState(false);
  const [dial, setDial] = useState('VDC20'); // OFF, VDC20, VAC, OHM, ADC
  const [reading, setReading] = useState('0.00');
  const [redProbe, setRedProbe] = useState({ x: 65, y: 45 }); // %
  const [blackProbe, setBlackProbe] = useState({ x: 35, y: 75 });
  const [isDragging, setIsDragging] = useState(null);

  // حساب القراءة بناء على وضع المجسات
  useEffect(() => {
    if (!isOn || dial === 'OFF') {
      setReading(' - - - ');
      return;
    }
    
    const interval = setInterval(() => {
      // محاكاة قراءة واقعية مثل الفيديو
      const baseVoltage = 9.0 + Math.random() * 1.5; // 9-10.5V
      
      // هل المجسات على نقاط صحيحة؟
      const redOnPositive = redProbe.x > 50 && redProbe.y < 60;
      const blackOnNegative = blackProbe.x < 50;
      
      if (dial === 'VDC20' && redOnPositive && blackOnNegative) {
        const val = (baseVoltage + (Math.random() - 0.5) * 0.2).toFixed(2);
        setReading(val);
      } else if (dial === 'VDC20') {
        // قراءة عشوائية مثل الفيديو 0.07
        if (Math.random() > 0.7) {
          setReading((Math.random() * 0.5).toFixed(2));
        } else {
          setReading('0.00');
        }
      } else {
        setReading('0.00');
      }
    }, 120);

    return () => clearInterval(interval);
  }, [isOn, dial, redProbe, blackProbe]);

  const handlePointerMove = (e, containerRect) => {
    if (!isDragging) return;
    const x = ((e.clientX || e.touches?.[0]?.clientX) - containerRect.left) / containerRect.width * 100;
    const y = ((e.clientY || e.touches?.[0]?.clientY) - containerRect.top) / containerRect.height * 100;
    const pos = { x: Math.max(5, Math.min(95, x)), y: Math.max(5, Math.min(95, y)) };
    
    if (isDragging === 'red') setRedProbe(pos);
    else setBlackProbe(pos);
  };

  return (
    <div className="w-full max-w-5xl mx-auto bg-[#eef2f7] rounded-2xl p-4 shadow-xl border" dir="rtl">
      <h2 className="text-xl font-bold mb-3 text-slate-800">🔬 محاكي الأفوميتر - جزء من المعمل الافتراضي</h2>
      
      {/* لوحة التحكم */}
      <div className="flex gap-2 mb-4 flex-wrap">
        <button 
          onClick={() => setIsOn(!isOn)}
          className={`px-4 py-2 rounded-lg font-bold ${isOn ? 'bg-green-500 text-white' : 'bg-slate-300'}`}
        >
          {isOn ? '● الدائرة شغالة' : '○ تشغيل الدائرة'}
        </button>
        <div className="flex gap-1 bg-white rounded-lg p-1">
          {['OFF','VDC20','VAC','OHM','ADC'].map(d => (
            <button key={d} onClick={() => setDial(d)}
              className={`px-3 py-1 rounded text-sm ${dial===d ? 'bg-yellow-400 font-bold' : 'bg-slate-100'}`}>
              {d}
            </button>
          ))}
        </div>
      </div>

      {/* منطقة المعمل */}
      <div 
        className="relative bg-white rounded-xl h-[420px] overflow-hidden border-2 border-slate-200 select-none touch-none"
        onPointerMove={(e) => handlePointerMove(e, e.currentTarget.getBoundingClientRect())}
        onPointerUp={() => setIsDragging(null)}
        onPointerLeave={() => setIsDragging(null)}
      >
        {/* البطارية */}
        <div className="absolute left-4 top-1/2 -translate-y-1/2 w-20 h-32 bg-slate-800 rounded-lg flex flex-col items-center justify-center border-2 border-slate-600">
          <div className="text-yellow-400 font-bold">9V</div>
          <div className="w-8 h-3 bg-slate-400 mt-2 rounded-sm"></div>
          <div className="text-[10px] text-white mt-2">بطارية</div>
        </div>

        {/* اللوحة */}
        <div className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 w-40 h-48 bg-green-600 rounded-lg border-4 border-green-700 shadow-inner">
          <div className="absolute -top-12 left-1/2 -translate-x-1/2">
            <div className={`w-8 h-10 rounded-full ${isOn ? 'bg-yellow-300 shadow-[0_0_20px_#fde047]' : 'bg-gray-300'} border-2 border-white transition-all`}></div>
            <div className="text-[10px] text-center mt-1 font-bold">لمبة</div>
          </div>
          <div className="grid grid-cols-4 gap-2 p-4 mt-6">
            {Array.from({length:12}).map((_,i)=>(
              <div key={i} className="w-3 h-3 bg-green-800 rounded-full border border-green-900"></div>
            ))}
          </div>
        </div>

        {/* الأفوميتر */}
        <div className="absolute right-4 top-1/2 -translate-y-1/2 w-44 bg-[#fde68a] rounded-2xl p-3 border-4 border-yellow-400 shadow-xl">
          <div className="bg-black text-green-400 font-mono text-2xl p-2 rounded text-center tracking-widest">
            {reading}
          </div>
          <div className="mt-3 w-16 h-16 bg-gray-300 rounded-full mx-auto border-4 border-gray-400 relative">
            <div className="absolute w-1 h-6 bg-black left-1/2 -translate-x-1/2 origin-bottom" style={{transform: `translateX(-50%) rotate(${dial==='VDC20'? '45deg' : dial==='OFF'? '-90deg' : '0deg'})`}}></div>
          </div>
          <div className="text-[10px] text-center mt-2">V Ω mA</div>
        </div>

        {/* المجسات */}
        <div 
          className="absolute w-8 h-8 bg-red-500 rounded-full border-2 border-white shadow-lg cursor-grab flex items-center justify-center text-white font-bold z-10"
          style={{ left: `${redProbe.x}%`, top: `${redProbe.y}%` }}
          onPointerDown={() => setIsDragging('red')}
        >R</div>
        <div 
          className="absolute w-8 h-8 bg-black rounded-full border-2 border-white shadow-lg cursor-grab flex items-center justify-center text-white font-bold z-10"
          style={{ left: `${blackProbe.x}%`, top: `${blackProbe.y}%` }}
          onPointerDown={() => setIsDragging('black')}
        >B</div>

        {/* أسلاك */}
        <svg className="absolute inset-0 w-full h-full pointer-events-none">
          <path d={`M ${redProbe.x}% ${redProbe.y}% Q 75% 30% 88% 50%`} stroke="red" strokeWidth="2" fill="none" />
          <path d={`M ${blackProbe.x}% ${blackProbe.y}% Q 40% 80% 88% 65%`} stroke="black" strokeWidth="2" fill="none" />
        </svg>
      </div>

      <p className="text-xs text-slate-500 mt-2">اسحب المجس الأحمر والأسود (R,B) إلى نقاط القياس على اللوحة الخضراء لقياس الجهد.</p>
    </div>
  );
}
